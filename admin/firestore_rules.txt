rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /device_bindings/{email} {
      // Admin can read/write ALL docs
      allow read, write: if request.auth != null 
                         && request.auth.token.email == 'goureesh3690@gmail.com';
      
      // Regular users can read their own doc
      allow read: if request.auth != null 
                  && request.auth.token.email == email;
      
      // Users can create their doc, but cannot set manualPremium
      allow create: if request.auth != null 
                    && request.auth.token.email == email
                    && !('manualPremium' in request.resource.data);
      
      // Users can update their doc, but cannot change protected admin fields
      allow update: if request.auth != null 
                    && request.auth.token.email == email
                    && (!('manualPremium' in request.resource.data) 
                        || request.resource.data.manualPremium == resource.data.manualPremium)
                    && (!('manualPremiumExpiry' in request.resource.data) 
                        || request.resource.data.manualPremiumExpiry == resource.data.manualPremiumExpiry)
                    && (!('muhurtaUnlocked' in request.resource.data) 
                        || request.resource.data.muhurtaUnlocked == resource.data.muhurtaUnlocked)
                    && (!('isStudent' in request.resource.data) 
                        || request.resource.data.isStudent == resource.data.isStudent)
                    && (!('blocked' in request.resource.data) 
                        || request.resource.data.blocked == resource.data.blocked);
    }

    match /installs/{deviceId} {
      allow read, write: if request.auth != null;
    }

    match /user_data/{email} {
      allow read, write: if request.auth != null 
                         && request.auth.token.email == email;
    }

    match /user_data/{email}/chunks/{chunkId} {
      allow read, write: if request.auth != null 
                         && request.auth.token.email == email;
    }

    match /appointments/{email} {
      allow read, write: if request.auth != null 
                         && request.auth.token.email == email;
    }

    match /appointments/{email}/requests/{requestId} {
      allow read, write: if request.auth != null 
                         && request.auth.token.email == email;
    }

    match /testers/{email} {
      allow read: if request.auth != null 
                  && request.auth.token.email == email;
      allow write: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
